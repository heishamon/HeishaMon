# Known issue: compressor stop and kick in the same tick. The restore (shift - bonus)
# and the kick (stale shift + 1) go out together; timer 11 then leaves the bonus in place
# while #Bonus is already 0.
# Preamble: compressor running at low load in heat-only mode, outlet on target.
silent Defrosting_State 0
silent Heat_Power_Consumption 400
silent Pump_Duty 80
silent Discharge_Temp 30
silent Compressor_Freq 25
silent Main_Outlet_Temp 33
silent Z1_Water_Target_Temp 33
silent Z1_Heat_Request_Temp 0
silent Operating_Mode_State 0
silent Heatpump_State 1
silent DHW_Temp 45
silent DHW_Target_Temp 45
silent DHW_Power_Consumption 0
boot
set Main_Outlet_Temp 35
adv 190
expectcmd SetZ1HeatRequestTemperature 1
pump
note compressor stops and the outlet falls below target - 3 K within one tick
set Heat_Power_Consumption 0
set Compressor_Freq 0
set Main_Outlet_Temp 29
adv 15
expectcmd SetZ1HeatRequestTemperature 0
expectcmd SetZ1HeatRequestTemperature 2
pump
adv 20
expectcmd SetZ1HeatRequestTemperature 1
pump
note the shift stays +1 with #Bonus = 0: nothing removes it
set Main_Outlet_Temp 31
adv 600
expectnone
