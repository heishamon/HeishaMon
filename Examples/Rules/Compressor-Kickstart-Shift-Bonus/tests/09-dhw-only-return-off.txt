# Heat pump off, something outside the rules starts a DHW-only run.
# At the end, the rules switch the heat pump off again and select heat-only.
# Known issue: the commands repeat every 15 s until the read-back changes.
# Preamble: compressor running at low load in heat-only mode, outlet on target.
silent Defrosting_State 0
silent Heat_Power_Consumption 400
silent Pump_Duty 80
silent Discharge_Temp 30
silent Compressor_Freq 25
silent Main_Outlet_Temp 33
silent Z1_Water_Target_Temp 33
silent Z1_Heat_Request_Temp 0
silent Operating_Mode_State 0
silent Heatpump_State 1
silent DHW_Temp 45
silent DHW_Target_Temp 45
silent DHW_Power_Consumption 0
boot
set Heatpump_State 0
set Pump_Duty 0
set Heat_Power_Consumption 0
set Compressor_Freq 0
set DHW_Temp 40
adv 40
expectnone
note external DHW-only start
set Operating_Mode_State 3
set Heatpump_State 1
set Compressor_Freq 30
set DHW_Power_Consumption 1500
adv 60
expectnone
note tank on target, DHW power 0
set DHW_Temp 45
set DHW_Power_Consumption 0
adv 15
expectcmd SetHeatpump 0
expectcmd SetOperationMode 0
note no read-back yet: the next tick sends both commands again
adv 15
expectcmd SetHeatpump 0
expectcmd SetOperationMode 0
set Heatpump_State 0
set Operating_Mode_State 0
adv 60
expectnone
