# Heatpump running (typical live setup, already heat-only mode): switch
# it off, boost through the fixed 3-minute wait / DHW-only / 10-minute
# run, then restore the heatpump afterwards.
silent Heatpump_State 1
silent Operating_Mode_State 0
silent DHW_Temp 40
silent Force_Heater_State 0
boot
adv 60
expectcmd SetHeatpump 0
set Heatpump_State 0
adv 60
expectcmd SetForceHeater 1
set Force_Heater_State 1
adv 180
expectcmd SetOperationMode 3
set Operating_Mode_State 3
adv 600
expectcmd SetOperationMode 0
expectcmd SetForceHeater 0
set Operating_Mode_State 0
set Force_Heater_State 0
adv 60
expectcmd SetHeatpump 1
set Heatpump_State 1
expectnone
